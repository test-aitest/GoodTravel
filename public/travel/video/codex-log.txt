+FFMPEG = '/opt/homebrew/bin/ffmpeg'
+FFPROBE = '/opt/homebrew/bin/ffprobe'
+WIDTH, HEIGHT = 480, 270
+FRAME_BYTES = WIDTH * HEIGHT
+
+def probe(path):
+    return json.loads(subprocess.check_output([
+        FFPROBE, '-v', 'error', '-count_frames', '-show_streams',
+        '-show_format', '-of', 'json', str(path)]))
+
+def faststart(path):
+    atoms = []
+    with path.open('rb') as f:
+        while header := f.read(8):
+            size, atom = struct.unpack('>I4s', header)
+            header_size = 8
+            if size == 1:
+                size = struct.unpack('>Q', f.read(8))[0]
+                header_size = 16
+            atoms.append(atom)
+            if size == 0:
+                break
+            f.seek(size - header_size, 1)
+    return atoms.index(b'moov') < atoms.index(b'mdat')
+
+report = []
+for name in ['rail-fuji', 'plane-clouds', 'kyoto']:
+    for extension in ['mp4', 'webm']:
+        path = ROOT / 'video' / f'{name}.{extension}'
+        data = probe(path)
+        stream = data['streams'][0]
+        assert len(data['streams']) == 1 and stream['codec_type'] == 'video'
+        assert stream['codec_name'] == {'mp4': 'h264', 'webm': 'vp9'}[extension]
+        assert (stream['width'], stream['height']) == (1920, 1080)

+            '-vf', f'scale={WIDTH}:{HEIGHT}', '-pix_fmt', 'gray',

+        frames = [memoryview(raw)[i:i+FRAME_BYTES] for i in range(0,len(raw),FRAME_BYTES)]
